     9→    /// <summary>
    10→    /// TIPOS DE NODOS (estilo Path of Exile):
    11→    /// - Small: bono menor (ej: +5 vida, +2 daño)
    12→    /// - Notable: bono mediano potente (ej: +15% daño de fuego, +10% velocidad)
    13→    /// - Keystone: cambia la build completamente (ej: lifesteal, regen de mana)
    14→    /// - Cluster: grupo de small+notable alrededor de un nodo central
    15→    /// - Ascendancy: potenciadores procedurales post-nivel 100
    16→    /// </summary>
    17→    public enum NodeType { Small, Notable, Keystone, Cluster, Ascendancy }

    19→    /// <summary>
    20→    /// Nodo del árbol de habilidades estilo PoE.
    21→    /// En PoE, los nodos forman un GRAFO DIRIGIDO (no radial).
    22→    /// Cada nodo tiene conexiones (aristas) a otros nodos.
    23→    /// El jugador debe seguir caminos desde su punto de partida.
    24→    /// </summary>
    25→    public class PoESkillNode
    26→    {
    27→        public string Id;
    28→        public string Name;
    29→        public string Branch;       // sub-rama temática
    30→        public NodeType Type;
    31→        public int Cost;             // puntos de habilidad
    32→        public string Effect;        // descripción del efecto
    33→        public List<string> Connections = new(); // IDs de nodos adyacentes
    34→        // Posición en el canvas (coordenadas absolutas, no normalizadas)
    35→        public float X;
    36→        public float Y;
    37→        // Tamaño visual según tipo
    38→        public float Radius => Type switch
    39→        {
    40→            NodeType.Small => 8f,
    41→            NodeType.Notable => 14f,
    42→            NodeType.Keystone => 22f,
    43→            NodeType.Cluster => 12f,
    44→            NodeType.Ascendancy => 18f,
    45→            _ => 8f,
    46→        };
    47→    }

   263→        // HELPER: Construir un cluster completo (estilo PoE)
   264→        // Cada cluster tiene: 3-5 small nodes → 1 notable → opcional keystone
   265→        // ====================================================================
   266→        private static void BuildCluster(
   267→            PoESkillTree tree, string clusterId, string clusterName, string icon,
   268→            string desc, bool hasKeystone, float angleDeg, float dist,
   269→            float cx, float cy, BranchType branch)
   270→        {
   271→            float rad = angleDeg * (float)Math.PI / 180f;
   272→            float clusterX = cx + (float)Math.Cos(rad) * dist;
   273→            float clusterY = cy + (float)Math.Sin(rad) * dist;
   274→
   275→            // Nodo de entrada del cluster (conectado al camino principal)
   276→            string entryId = $"{clusterId}-entry";
   277→            var entryNode = new PoESkillNode
   278→            {
   279→                Id = entryId, Name = $"{clusterName} (Entrada)", Branch = clusterName,
   280→                Type = NodeType.Small, Cost = 1,
   281→                Effect = GetSmallEffect(clusterId, branch, 0),
   282→                X = clusterX - 40f, Y = clusterY,
   283→            };
   284→            tree.Nodes.Add(entryNode);
   285→
   286→            // 3 small nodes en línea
   287→            string prevId = entryId;
   288→            for (int i = 0; i < 3; i++)
   289→            {
   290→                string nodeId = $"{clusterId}-small-{i}";
   291→                float nx = clusterX + (i - 1) * 40f;
   292→                float ny = clusterY + 30f;
   293→                var node = new PoESkillNode
   294→                {
   295→                    Id = nodeId, Name = $"{clusterName} (Nodo {i+1})", Branch = clusterName,
   296→                    Type = NodeType.Small, Cost = 1,
   297→                    Effect = GetSmallEffect(clusterId, branch, i + 1),
   298→                    X = nx, Y = ny,
   299→                };
   300→                node.Connections.Add(prevId);
   301→                tree.Nodes.Find(n => n.Id == prevId)!.Connections.Add(nodeId);
   302→                tree.Nodes.Add(node);
   303→                prevId = nodeId;
   304→            }
   305→
   306→            // Notable del cluster
   307→            string notableId = $"{clusterId}-notable";
   308→            var notable = new PoESkillNode
   309→            {
   310→                Id = notableId, Name = GetNotableName(clusterId, branch),
   311→                Branch = clusterName, Type = NodeType.Notable, Cost = 2,
   312→                Effect = GetNotableEffect(clusterId, branch),
   313→                X = clusterX + 20f, Y = clusterY + 80f,
   314→            };
   315→            notable.Connections.Add(prevId);
   316→            tree.Nodes.Find(n => n.Id == prevId)!.Connections.Add(notableId);
   317→            tree.Nodes.Add(notable);
   318→
   319→            // Keystone del cluster (opcional)
   320→            if (hasKeystone)
   321→            {
   322→                string keystoneId = $"{clusterId}-keystone";
   323→                var keystone = new PoESkillNode
   324→                {
   325→                    Id = keystoneId, Name = GetKeystoneName(clusterId, branch),
   326→                    Branch = clusterName, Type = NodeType.Keystone, Cost = 3,
   327→                    Effect = GetKeystoneEffect(clusterId, branch),
   328→                    X = clusterX + 60f, Y = clusterY + 130f,
   329→                };
   330→                keystone.Connections.Add(notableId);
   331→                tree.Nodes.Find(n => n.Id == notableId)!.Connections.Add(keystoneId);
   332→                tree.Nodes.Add(keystone);
   333→            }
   334→        }
   335→
   336→        // Conectar dos clusters (camino entre ellos)
   337→        private static void ConnectClusters(PoESkillTree tree, string fromCluster, string toCluster)
   338→        {
   339→            string fromId = $"{fromCluster}-entry";
   340→            string toId = $"{toCluster}-entry";
   341→            var from = tree.Nodes.Find(n => n.Id == fromId);
   342→            var to = tree.Nodes.Find(n => n.Id == toId);
   343→            if (from != null && to != null)
   344→            {
   345→                from.Connections.Add(toId);
   346→                to.Connections.Add(fromId);
   347→            }
   348→        }

   352→        // ====================================================================
   353→        private static void BuildAscendancyZone(PoESkillTree tree, float cx, float cy, BranchType branch)
   354→        {
   355→            // Nodo de entrada a la zona de Ascendancy
   356→            string ascendEntryId = "ascend-entry";
   357→            var ascendEntry = new PoESkillNode
   358→            {
   359→                Id = ascendEntryId, Name = "Ascendencia (Lv 100+)", Branch = "Ascendancy",
   360→                Type = NodeType.Keystone, Cost = 0,
   361→                Effect = "Desbloquea potenciadores procedurales. Genera nuevos nodos cada nivel.",
   362→                X = cx, Y = cy,
   363→            };
   364→            tree.Nodes.Add(ascendEntry);
   365→            // Conectar al nodo inicial
   366→            tree.Nodes.Find(n => n.Id == "start")!.Connections.Add(ascendEntryId);
   367→            ascendEntry.Connections.Add("start");
   368→
   369→            // 6 potenciadores procedurales base (se generan más al subir de nivel)
   370→            string[] ascendNames = {
   371→                "Potenciador Estelar I", "Potenciador Estelar II", "Potenciador Estelar III",
   372→                "Potenciador Cósmico I", "Potenciador Cósmico II", "Potenciador Cósmico III",
   373→            };
   374→            string[] ascendEffects = GetAscendancyEffects(branch);
   375→
   376→            string prevAscendId = ascendEntryId;
   377→            for (int i = 0; i < 6; i++)
   378→            {
   379→                float angle = i * 60f * (float)Math.PI / 180f;
   380→                string nodeId = $"ascend-{i}";
   381→                var node = new PoESkillNode
   382→                {
   383→                    Id = nodeId, Name = ascendNames[i], Branch = "Ascendancy",
   384→                    Type = NodeType.Ascendancy, Cost = 5,
   385→                    Effect = ascendEffects[i],
   386→                    X = cx + (float)Math.Cos(angle) * 200f,
   387→                    Y = cy + (float)Math.Sin(angle) * 200f,
   388→                };
   389→                node.Connections.Add(prevAscendId);
   390→                tree.Nodes.Find(n => n.Id == prevAscendId)!.Connections.Add(nodeId);
   391→                tree.Nodes.Add(node);
   392→                prevAscendId = nodeId;
   393→            }
   394→
   395→            // KEYSTONE DE REGENERACIÓN DE SALUD (nivel 100+)
   396→            // Distancia y Melee: regenera 0.01% del daño causado como salud
   397→            // Magia: igual pero las invocaciones NO cuentan
   398→            string regenKeystoneId = "ascend-regen-keystone";
   399→            string regenEffect = branch == BranchType.Magic
   400→                ? "REGENERACIÓN VITAL: Regenera el 0.01% de tu salud máxima por cada punto de daño mágico causado. El daño de invocaciones NO cuenta para esta regeneración."
   401→                : "REGENERACIÓN VITAL: Regenera el 0.01% de tu salud máxima por cada punto de daño causado al enemigo. Lifesteal pasivo permanente.";
   402→
   403→            var regenKeystone = new PoESkillNode
   404→            {
   405→                Id = regenKeystoneId, Name = "Regeneración Vital (Keystone)",
   406→                Branch = "Ascendancy", Type = NodeType.Keystone, Cost = 10,
   407→                Effect = regenEffect,
   408→                X = cx + 100f, Y = cy + 250f,
   409→            };
   410→            regenKeystone.Connections.Add(prevAscendId);
   411→            tree.Nodes.Find(n => n.Id == prevAscendId)!.Connections.Add(regenKeystoneId);
   412→            tree.Nodes.Add(regenKeystone);
   413→
   414→            // Potenciadores adicionales que se generan proceduralmente
   415→            // (el sistema generará más nodos en runtime cuando el jugador suba de nivel post-100)
   416→            for (int i = 6; i < 12; i++)
   417→            {
   418→                float angle = i * 60f * (float)Math.PI / 180f;
   419→                string nodeId = $"ascend-proc-{i}";
   420→                var node = new PoESkillNode
   421→                {
   422→                    Id = nodeId, Name = $"Potenciador Procedural {i-5}",
   423→                    Branch = "Ascendancy", Type = NodeType.Ascendancy, Cost = 5,
   424→                    Effect = GetProceduralEffect(branch, i),
   425→                    X = cx + (float)Math.Cos(angle) * 300f,
   426→                    Y = cy + (float)Math.Sin(angle) * 300f,
   427→                };
   428→                node.Connections.Add(prevAscendId);
   429→                tree.Nodes.Find(n => n.Id == prevAscendId)!.Connections.Add(nodeId);
   430→                tree.Nodes.Add(node);
   431→                prevAscendId = nodeId;
   432→            }
   433→        }

   446→                ("celestial", _) => $"+{5 + index * 3}% daño solar",
   447→                ("phantom", _) => $"+{3 + index}% probabilidad de fase",
   448→                ("velocity", _) => $"+{5 + index * 2}% velocidad de movimiento",
   449→                ("piercing", _) => $"+1 perforación de enemigos",
   450→                ("elemental", _) => $"+{5 + index * 3}% daño elemental",
   451→                ("range", _) => $"+{10 + index * 5} tiles de alcance",
   452→                ("ammo", _) => $"+{5 + index * 2}% probabilidad de no consumir munición",
   453→                ("survival", _) => $"+{10 + index * 5} vida máxima",
   454→                ("absorption", _) => $"+1 slot de runa de memoria",
   455→
