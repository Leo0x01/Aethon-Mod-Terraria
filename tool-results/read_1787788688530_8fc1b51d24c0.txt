    91→    91→        private void InitStars()
    92→    92→        {
    93→    93→            if (_stars != null) return;
    94→    94→            var rand = new Random(42); // seed fijo para consistencia
    95→    95→            _stars = new Star[200];
    96→    96→            for (int i = 0; i < _stars.Length; i++)
    97→    97→            {
    98→    98→                int layer = rand.Next(3); // 0, 1, 2
    99→    99→                _stars[i] = new Star
   100→   100→                {
   101→   101→                    Pos = new Vector2((float)rand.NextDouble(), (float)rand.NextDouble()),
   102→   102→                    Size = 1f + (float)rand.NextDouble() * (layer == 2 ? 2.5f : layer == 1 ? 1.8f : 1.2f),
   103→   103→                    Twinkle = 0.5f + (float)rand.NextDouble() * 2f,
   104→   104→                    Phase = (float)rand.NextDouble() * MathF.PI * 2f,
   105→   105→                    Layer = layer,
   106→   106→                    Color = layer == 2
   107→   107→                        ? new Color(245, 196, 81)   // dorado (cerca)
   108→   108→                        : layer == 1
   109→   109→                            ? new Color(179, 136, 255) // violeta (medio)
   110→   110→                            : new Color(200, 220, 255), // blanco-azul (lejos)
   111→   111→                };

   115→   115→        public void BuildPoETree(PoESkillTree tree)
   116→   116→        {
   117→   117→            CurrentTree = tree;
   118→   118→            _allNodes = tree.Nodes ?? new List<PoESkillNode>();
   119→   119→            _allConnections.Clear();
   120→   120→            _nodeIndex.Clear();
   121→   121→            for (int i = 0; i < _allNodes.Count; i++)
   122→   122→                _nodeIndex[_allNodes[i].Id] = i;
   123→   123→            foreach (var node in _allNodes)
   124→   124→            {
   125→   125→                if (!_nodeIndex.TryGetValue(node.Id, out int fromIdx)) continue;
   126→   126→                foreach (var connId in node.Connections)
   127→   127→                    if (_nodeIndex.TryGetValue(connId, out int j))
   128→   128→                        _allConnections.Add((fromIdx, j));
   129→   129→            }
   130→   130→            if (_allNodes.Count == 0)
   131→   131→            {
   132→   132→                _minX = _maxX = _minY = _maxY = 0;
   133→   133→                _rangeX = _rangeY = 1;
   134→   134→                return;
   135→   135→            }
   136→   136→            _minX = float.MaxValue; _maxX = float.MinValue;
   137→   137→            _minY = float.MaxValue; _maxY = float.MinValue;
   138→   138→            foreach (var node in _allNodes)
   139→   139→            {
   140→   140→                _minX = Math.Min(_minX, node.X);
   141→   141→                _maxX = Math.Max(_maxX, node.X);
   142→   142→                _minY = Math.Min(_minY, node.Y);
   143→   143→                _maxY = Math.Max(_maxY, node.Y);
   144→   144→            }
   145→   145→            _rangeX = Math.Max(1, _maxX - _minX + 1);
   146→   146→            _rangeY = Math.Max(1, _maxY - _minY + 1);
   147→   147→        }

   185→   185→        private Vector2 NodeToScreen(PoESkillNode node)
   186→   186→        {
   187→   187→            if (_rangeX <= 0 || _rangeY <= 0) return new Vector2(Main.screenWidth / 2f, Main.screenHeight / 2f);
   188→   188→            float cx = Main.screenWidth / 2f;
   189→   189→            float cy = Main.screenHeight / 2f;
   190→   190→            float scale = Math.Min(Main.screenWidth / _rangeX, Main.screenHeight / _rangeY) * 0.45f * _zoom;
   191→   191→            return new Vector2(
   192→   192→                (node.X - _minX) * scale - (_rangeX * scale / 2f) + cx + _panOffset.X,
   193→   193→                (node.Y - _minY) * scale - (_rangeY * scale / 2f) + cy + _panOffset.Y);
   194→   194→        }

   196→   196→        private PoESkillNode? FindHoveredNode()
   197→   197→        {
   198→   198→            Vector2 mouse = new(Main.mouseX, Main.mouseY);
   199→   199→            PoESkillNode? closest = null;
   200→   200→            float closestDist = float.MaxValue;
   201→   201→            foreach (var node in _allNodes)
   202→   202→            {
   203→   203→                Vector2 pos = NodeToScreen(node);
   204→   204→                float dist = Vector2.Distance(pos, mouse);
   205→   205→                float radius = node.Radius * 2.2f * _zoom + 8f;
   206→   206→                if (dist < radius && dist < closestDist) { closestDist = dist; closest = node; }
   207→   207→            }
   208→   208→            return closest;
   209→   209→        }

   211→   211→        public void Draw()
   212→   212→        {
   213→   213→            if (!IsVisible) return;
   214→   214→            var sp = Main.LocalPlayer?.GetModPlayer<ShardPlayer>();
   215→   215→            if (sp == null) return;
   216→   216→

   220→   220→            try
   221→   221→            {
   222→   222→                DrawBackground(sb);
   223→   223→                DrawStars(sb);
   224→   224→                UpdateMeteors();
   225→   225→                DrawMeteors(sb);

   227→   227→                if (CurrentTree == null || _allNodes.Count == 0)
   228→   228→                {

   233→   233→                    DrawBottomHelp(sb);

   235→   235→                    return;

   237→   237→
   238→   238→                DrawConnections(sb, sp);
   239→   239→                _hoveredNode = FindHoveredNode();
   240→   240→                _hoverAlpha = MathHelper.Lerp(_hoverAlpha, _hoveredNode != null ? 1f : 0f, 0.2f);
   241→   241→                DrawNodes(sb, sp);
   242→   242→                DrawTooltip(sb, sp);
   243→   243→                DrawTopBar(sb, sp, sp.CumulativeSkillPoints(), sp.AllocatedNodes.Count);

   245→   245→                HandleInput(sp);
   246→   246→            }
   247→   247→            catch (Exception ex)
   248→   248→            {
   249→   249→                // Nunca propagar excepciones desde Draw — podria romper el render loop.
   250→   250→                ModContent.GetInstance<AethonMod>()?.Logger?.Error("SkillTreeUI.Draw error", ex);
   251→   251→                IsVisible = false;
   252→   252→            }
   253→   253→        }

   269→   269→
   270→   270→            // Nebulosa central pulsante
   271→   271→            float pulse = 0.85f + (float)Math.Sin(_time * 0.5) * 0.15f;
   272→   272→            DrawRadial(sb, 0.5f, 0.5f, 0.3f * pulse, new Color(80, 40, 160, 18));
   273→   273→

   275→   275→            for (int r = 320; r > 0; r -= 20)

   278→   278→                sb.Draw(TextureAssets.MagicPixel.Value,
   279→   279→                    new Rectangle(Main.screenWidth / 2 - r, Main.screenHeight / 2 - r, r * 2, r * 2),

   284→   284→        private void DrawStars(SpriteBatch sb)
   285→   285→        {
   286→   286→            if (_stars == null) return;
   287→   287→            for (int i = 0; i < _stars.Length; i++)
   288→   288→            {
   289→   289→                var s = _stars[i];
   290→   290→                // Parallax: las estrellas cercanas (layer 2) se mueven mas con el pan
   291→   291→                float parallax = s.Layer == 2 ? 0.4f : s.Layer == 1 ? 0.2f : 0.08f;
   292→   292→                float sx = (s.Pos.X * Main.screenWidth + _panOffset.X * parallax) % Main.screenWidth;
   293→   293→                float sy = (s.Pos.Y * Main.screenHeight + _panOffset.Y * parallax) % Main.screenHeight;
   294→   294→                if (sx < 0) sx += Main.screenWidth;
   295→   295→                if (sy < 0) sy += Main.screenHeight;
   296→   296→
   297→   297→                // Twinkle (parpadeo)
   298→   298→                float twinkle = 0.6f + 0.4f * (float)Math.Sin(_time * s.Twinkle + s.Phase);
   299→   299→                int alpha = (int)(255 * twinkle * (s.Layer == 2 ? 0.95f : s.Layer == 1 ? 0.7f : 0.45f));
   300→   300→                alpha = MathHelper.Clamp(alpha, 0, 255);
   301→   301→                Color c = new Color(s.Color.R, s.Color.G, s.Color.B, alpha);
   302→   302→
   303→   303→                // Estrella con glow para las cercanas
   304→   304→                if (s.Layer == 2 && s.Size > 2f)
   305→   305→                {
   306→   306→                    sb.Draw(TextureAssets.MagicPixel.Value,
   307→   307→                        new Rectangle((int)sx - 2, (int)sy - 2, 5, 5),
   308→   308→                        new Color(c.R, c.G, c.B, alpha / 4));
   309→   309→                }
   310→   310→                sb.Draw(TextureAssets.MagicPixel.Value,
   311→   311→                    new Rectangle((int)sx, (int)sy, (int)s.Size, (int)s.Size), c);
   312→   312→            }
   313→   313→        }

   315→   315→        private void UpdateMeteors()
   316→   316→        {
   317→   317→            _meteorTimer++;
   318→   318→            if (_meteorTimer > 180 + Main.rand.Next(120) && _meteors.Count < 3)
   319→   319→            {
   320→   320→                _meteorTimer = 0;
   321→   321→                var meteor = new Meteor
   322→   322→                {
   323→   323→                    Pos = new Vector2(Main.rand.NextFloat(Main.screenWidth), -20),
   324→   324→                    Vel = new Vector2(Main.rand.NextFloat(-2, 2), Main.rand.NextFloat(4, 8)),
   325→   325→                    Life = 0,
   326→   326→                    MaxLife = 60 + Main.rand.Next(40),
   327→   327→                    Color = Main.rand.NextBool() ? new Color(245, 196, 81) : new Color(179, 136, 255),
   328→   328→                };
   329→   329→                _meteors.Add(meteor);
   330→   330→            }
   331→   331→
   332→   332→            for (int i = _meteors.Count - 1; i >= 0; i--)
   333→   333→            {
   334→   334→                var m = _meteors[i];
   335→   335→                m.Pos += m.Vel;
   336→   336→                m.Life++;
   337→   337→                _meteors[i] = m;
   338→   338→                if (m.Life > m.MaxLife || m.Pos.Y > Main.screenHeight + 50)
   339→   339→                    _meteors.RemoveAt(i);
   340→   340→            }
   341→   341→        }

   343→   343→        private void DrawMeteors(SpriteBatch sb)
   344→   344→        {
   345→   345→            foreach (var m in _meteors)
   346→   346→            {
   347→   347→                float fade = 1f;
   348→   348→                if (m.Life < 5) fade = m.Life / 5f;
   349→   349→                else if (m.Life > m.MaxLife - 10) fade = (m.MaxLife - m.Life) / 10f;
   350→   350→                fade = MathHelper.Clamp(fade, 0, 1);
   351→   351→                // Cola del meteoro
   352→   352→                for (int i = 0; i < 8; i++)
   353→   353→                {
   354→   354→                    var trailPos = m.Pos - m.Vel * i * 1.5f;
   355→   355→                    int a = (int)(fade * (255 - i * 28));
   356→   356→                    if (a < 0) a = 0;
   357→   357→                    sb.Draw(TextureAssets.MagicPixel.Value,
   358→   358→                        new Rectangle((int)trailPos.X - 1, (int)trailPos.Y - 1, 3, 3),
   359→   359→                        new Color(m.Color.R, m.Color.G, m.Color.B, a));
   360→   360→                }
   361→   361→                // Cabeza
   362→   362→                sb.Draw(TextureAssets.MagicPixel.Value,
   363→   363→                    new Rectangle((int)m.Pos.X - 2, (int)m.Pos.Y - 2, 4, 4),
   364→   364→                    new Color(m.Color.R, m.Color.G, m.Color.B, (int)(fade * 255)));
   365→   365→            }
   366→   366→        }

   371→   371→        private void DrawConnections(SpriteBatch sb, ShardPlayer sp)
   372→   372→        {
   373→   373→            foreach (var (fromIdx, toIdx) in _allConnections)
   374→   374→            {
   375→   375→                if (fromIdx < 0 || fromIdx >= _allNodes.Count || toIdx < 0 || toIdx >= _allNodes.Count) continue;
   376→   376→                Vector2 from = NodeToScreen(_allNodes[fromIdx]);
   377→   377→                Vector2 to = NodeToScreen(_allNodes[toIdx]);
   378→   378→                bool bothActive = sp.AllocatedNodes.Contains(_allNodes[fromIdx].Id) && sp.AllocatedNodes.Contains(_allNodes[toIdx].Id);
   379→   379→                bool oneActive = sp.AllocatedNodes.Contains(_allNodes[fromIdx].Id) || sp.AllocatedNodes.Contains(_allNodes[toIdx].Id);
   380→   380→
   381→   381→                Color lc;
   382→   382→                float th;
   383→   383→                if (bothActive) { lc = new Color(245, 196, 81, 230); th = 3.5f; }
   384→   384→                else if (oneActive) { lc = new Color(180, 140, 80, 160); th = 2.2f; }
   385→   385→                else { lc = new Color(60, 50, 90, 90); th = 1.5f; }
   386→   386→
   387→   387→                DrawLine(sb, from, to, lc, th);
   388→   388→            }
   389→   389→        }

   391→   391→        private void DrawNodes(SpriteBatch sb, ShardPlayer sp)
   392→   392→        {
   393→   393→            foreach (var node in _allNodes)
   394→   394→            {
   395→   395→                Vector2 pos = NodeToScreen(node);
   396→   396→                bool allocated = sp.AllocatedNodes.Contains(node.Id);
   397→   397→                bool canAlloc = CanAllocate(node, sp);
   398→   398→                bool isHover = _hoveredNode != null && _hoveredNode.Id == node.Id;
   399→   399→                DrawNode(sb, pos, node, allocated, canAlloc, isHover);
   400→   400→            }
   401→   401→        }

   403→   403→        private void DrawTooltip(SpriteBatch sb, ShardPlayer sp)
   404→   404→        {
   405→   405→            if (_hoveredNode == null) return;
   406→   406→            var node = _hoveredNode;
   407→   407→            if (node == null) return;
   408→   408→            bool alloc = sp.AllocatedNodes.Contains(node.Id);
   409→   409→
   410→   410→            string typeStr = node.Type switch
   411→   411→            {
   412→   412→                NodeType.Small => "[Menor]",
   413→   413→                NodeType.Notable => "[Notable]",
   414→   414→                NodeType.Keystone => "[KEYSTONE]",
   415→   415→                NodeType.Ascendancy => "[Ascendencia]",
   416→   416→                NodeType.Cluster => "[Cluster]",
   417→   417→                _ => "",
   418→   418→            };
   419→   419→
   420→   420→            // Tooltip con fondo
   421→   421→            string[] lines = {
   422→   422→                $"{typeStr} {node.Name}",
   423→   423→                node.Effect ?? "",
   424→   424→                $"Rama: {node.Branch}",
   425→   425→                $"Coste: {node.Cost} pts",
   426→   426→                alloc ? "(Asignado — click para quitar)" : (CanAllocate(node, sp) ? "(Click para asignar)" : "(Requiere nodo adyacente)"),
   427→   427→            };
   428→   428→
   429→   429→            // Medir el tooltip
   430→   430→            float maxW = 0;
   431→   431→            foreach (var line in lines)
   432→   432→            {
   433→   433→                var size = FontAssets.MouseText.Value.MeasureString(line);

   559→   559→        private void HandleInput(ShardPlayer sp)
   560→   560→        {
   561→   561→            // Pan con click derecho
   562→   562→            if (Main.mouseRight)
   563→   563→            {
   564→   564→                Vector2 curMouse = new(Main.mouseX, Main.mouseY);
   565→   565→                if (!_isDragging)
   566→   566→                {
   567→   567→                    _isDragging = true;
   568→   568→                    _dragStart = curMouse;
   569→   569→                    _lastMouse = curMouse;
   570→   570→                }
   571→   571→                else
   572→   572→                {
   573→   573→                    _panOffset += curMouse - _lastMouse;
   574→   574→                    _lastMouse = curMouse;
   575→   575→                }
   576→   576→            }
   577→   577→            else _isDragging = false;
   578→   578→
   579→   579→            // Zoom con rueda
   580→   580→            int scroll = Main.mouseScroll;
   581→   581→            if (scroll != 0)
   582→   582→            {
   583→   583→                float zoomDelta = scroll > 0 ? 0.1f : -0.1f;
   584→   584→                _zoom = MathHelper.Clamp(_zoom + zoomDelta, 0.4f, 2.5f);
   585→   585→            }
   586→   586→
   587→   587→            // Click izquierdo para asignar/quitar
   588→   588→            if (Main.mouseLeft && !_mouseLeftPressed && _hoveredNode != null && !_isDragging)
   589→   589→            {
   590→   590→                _mouseLeftPressed = true;
   591→   591→                var node = _hoveredNode;
   592→   592→                if (node == null) return;
   593→   593→                if (sp.AllocatedNodes.Contains(node.Id))
   594→   594→                {
   595→   595→                    sp.AllocatedNodes.Remove(node.Id);
   596→   596→                    Terraria.Audio.SoundEngine.PlaySound(Terraria.ID.SoundID.MenuClose);
   597→   597→                    // Particulas al quitar
   598→   598→                    for (int i = 0; i < 8; i++)
   599→   599→                        Dust.NewDustPerfect(Main.LocalPlayer.Center, Terraria.ID.DustID.PurpleTorch,
   600→   600→                            new Vector2(Main.rand.NextFloat(-3, 3), Main.rand.NextFloat(-3, 3)),
   601→   601→                            100, new Color(179, 136, 255), 1f);
   602→   602→                }
   603→   603→                else if (CanAllocate(node, sp))
   604→   604→                {
   605→   605→                    int avail = sp.CumulativeSkillPoints() - sp.AllocatedNodes.Count;
   606→   606→                    if (avail < node.Cost)
   607→   607→                    {
   608→   608→                        Main.NewText($"No tienes suficientes puntos. Necesitas {node.Cost}, tienes {avail}.",
   609→   609→                            new Color(255, 120, 120));
   610→   610→                        Terraria.Audio.SoundEngine.PlaySound(Terraria.ID.SoundID.MenuClose);
   611→   611→                    }
   612→   612→                    else
   613→   613→                    {
   614→   614→                        sp.AllocatedNodes.Add(node.Id);
   615→   615→                        Terraria.Audio.SoundEngine.PlaySound(Terraria.ID.SoundID.MenuTick);
   616→   616→                        // Particulas de asignacion
   617→   617→                        Vector2 nodePos = NodeToScreen(node);
   618→   618→                        for (int i = 0; i < 16; i++)
   619→   619→                            Dust.NewDustPerfect(nodePos, Terraria.ID.DustID.GoldFlame,
   620→   620→                                new Vector2(Main.rand.NextFloat(-4, 4), Main.rand.NextFloat(-4, 4)),
   621→   621→                                100, new Color(245, 196, 81), 1.2f);
   622→   622→                    }
   623→   623→                }
   624→   624→            }
   625→   625→            if (!Main.mouseLeft) _mouseLeftPressed = false;
   626→   626→
   627→   627→            // Cerrar con Escape o K
   628→   628→            var kb = Main.keyState;
   629→   629→            var oldKb = Main.oldKeyState;
   630→   630→            if (kb.IsKeyDown(Keys.Escape) && !oldKb.IsKeyDown(Keys.Escape)) Hide();
   631→   631→
   632→   632→            var config = ModContent.GetInstance<Content.AethonConfig>();
   633→   633→            if (config != null && kb.IsKeyDown(config.SkillTreeKey) && !oldKb.IsKeyDown(config.SkillTreeKey))
   634→   634→                Hide();
   635→   635→
   636→   636→            // Bloquear el input del juego del mouse y teclas mientras la UI este abierta
   637→   637→            // (evita que el jugador ataque/mueva mientras asigna habilidades)
   638→   638→            Main.mouseLeft = false;
   639→   639→            Main.mouseRight = false;
   640→   640→            Main.mouseScroll = 0;
   641→   641→        }

   643→   643→        private bool CanAllocate(PoESkillNode node, ShardPlayer sp)
   644→   644→        {
   645→   645→            if (sp.AllocatedNodes.Contains(node.Id)) return false;
   646→   646→            if (node.Id == "start" || node.Cost == 0) return true;
   647→   647→            foreach (var aid in sp.AllocatedNodes)
   648→   648→                if (node.Connections.Contains(aid)) return true;
   649→   649→            return false;
   650→   650→        }

   670→   670→        private void DrawNode(SpriteBatch sb, Vector2 pos, PoESkillNode node, bool allocated, bool canAlloc, bool hovered)
