   465→    // ================================================================
   466→    //  HELPER: DrawAdditive — dibuja una textura con additive blending
   467→    // ================================================================
   468→    public static class VFXHelper
   469→    {
   470→        public static void DrawAdditive(string texturePath, Vector2 center, float scale, Color color, float rotation)
   471→        {
   472→            try
   473→            {
   474→                Texture2D tex = ModContent.Request<Texture2D>(texturePath).Value;
   475→                if (tex == null) return;
   476→                Vector2 origin = new Vector2(tex.Width / 2f, tex.Height / 2f);
   477→                Vector2 drawPos = center - Main.screenPosition;
   478→                Main.spriteBatch.End();
   479→                Main.spriteBatch.Begin(SpriteSortMode.Immediate, BlendState.Additive);
   480→                Main.spriteBatch.Draw(tex, drawPos, null, color, rotation, origin, scale, SpriteEffects.None, 0f);
   481→                Main.spriteBatch.End();
   482→                Main.spriteBatch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend);
   483→            }
   484→            catch { }
   485→        }
   486→    }
