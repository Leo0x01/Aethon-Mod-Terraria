    92→        public static PoESkillTree GetTree(BranchType branch)
    93→        {
    94→            lock (_lock)
    95→            {
    96→                return branch switch
    97→                {
    98→                    BranchType.Distance => _distanceTree ??= BuildDistanceTree(),
    99→                    BranchType.Melee => _meleeTree ??= BuildMeleeTree(),
   100→                    BranchType.Magic => _magicTree ??= BuildMagicTree(),
   101→                    _ => new PoESkillTree { Branch = branch },
   102→                };
   103→            }
   104→        }

   273→        // HELPER: Construir un cluster completo (estilo PoE)
   274→        // Cada cluster tiene: 3-5 small nodes → 1 notable → opcional keystone
   275→        // ====================================================================
   276→        private static void BuildCluster(
   277→            PoESkillTree tree, string clusterId, string clusterName, string icon,
   278→            string desc, bool hasKeystone, float angleDeg, float dist,
   279→            float cx, float cy, BranchType branch)
   280→        {
   281→            float rad = angleDeg * (float)Math.PI / 180f;
   282→            float clusterX = cx + (float)Math.Cos(rad) * dist;
   283→            float clusterY = cy + (float)Math.Sin(rad) * dist;
   284→
   285→            // Helper local para añadir bidireccional (con null check)
   286→            void ConnectBidirectional(string fromId, string toId)
   287→            {
   288→                var fromNode = tree.Nodes.Find(n => n.Id == fromId);
   289→                var toNode = tree.Nodes.Find(n => n.Id == toId);
   290→                if (fromNode != null && toNode != null)
   291→                {
   292→                    if (!fromNode.Connections.Contains(toId)) fromNode.Connections.Add(toId);
   293→                    if (!toNode.Connections.Contains(fromId)) toNode.Connections.Add(fromId);
   294→                }
   295→            }
   296→
   297→            // Nodo de entrada del cluster (conectado al camino principal)
   298→            string entryId = $"{clusterId}-entry";
   299→            var entryNode = new PoESkillNode
   300→            {
   301→                Id = entryId, Name = $"{clusterName} (Entrada)", Branch = clusterName,
   302→                Type = NodeType.Small, Cost = 1,
   303→                Effect = GetSmallEffect(clusterId, branch, 0),
   304→                X = clusterX - 40f, Y = clusterY,
   305→            };
   306→            tree.Nodes.Add(entryNode);
   307→
   308→            // 3 small nodes en línea
   309→            string prevId = entryId;
   310→            for (int i = 0; i < 3; i++)
   311→            {
   312→                string nodeId = $"{clusterId}-small-{i}";
   313→                float nx = clusterX + (i - 1) * 40f;
   314→                float ny = clusterY + 30f;
   315→                var node = new PoESkillNode
   316→                {
   317→                    Id = nodeId, Name = $"{clusterName} (Nodo {i+1})", Branch = clusterName,
   318→                    Type = NodeType.Small, Cost = 1,
   319→                    Effect = GetSmallEffect(clusterId, branch, i + 1),
   320→                    X = nx, Y = ny,
   321→                };
   322→                node.Connections.Add(prevId);
   323→                ConnectBidirectional(prevId, nodeId);
   324→                tree.Nodes.Add(node);
   325→                prevId = nodeId;
   326→            }
   327→
   328→            // Notable del cluster
   329→            string notableId = $"{clusterId}-notable";
   330→            var notable = new PoESkillNode
   331→            {
   332→                Id = notableId, Name = GetNotableName(clusterId, branch),
   333→                Branch = clusterName, Type = NodeType.Notable, Cost = 2,
   334→                Effect = GetNotableEffect(clusterId, branch),
   335→                X = clusterX + 20f, Y = clusterY + 80f,
   336→            };
   337→            notable.Connections.Add(prevId);
   338→            ConnectBidirectional(prevId, notableId);
   339→            tree.Nodes.Add(notable);
   340→
   341→            // Keystone del cluster (opcional)
   342→            if (hasKeystone)
   343→            {
   344→                string keystoneId = $"{clusterId}-keystone";
   345→                var keystone = new PoESkillNode
   346→                {
   347→                    Id = keystoneId, Name = GetKeystoneName(clusterId, branch),
   348→                    Branch = clusterName, Type = NodeType.Keystone, Cost = 3,
   349→                    Effect = GetKeystoneEffect(clusterId, branch),
   350→                    X = clusterX + 60f, Y = clusterY + 130f,
   351→                };
   352→                keystone.Connections.Add(notableId);
   353→                ConnectBidirectional(notableId, keystoneId);
   354→                tree.Nodes.Add(keystone);
   355→            }
   356→        }

   358→        // Conectar dos clusters (camino entre ellos)
   359→        private static void ConnectClusters(PoESkillTree tree, string fromCluster, string toCluster)
   360→        {
   361→            string fromId = $"{fromCluster}-entry";
   362→            string toId = $"{toCluster}-entry";
   363→            ConnectBidirectionalTree(tree, fromId, toId);
   364→        }
   365→
   366→        /// <summary>Conecta dos nodos bidireccionalmente (null-safe, evita duplicados).</summary>
   367→        private static void ConnectBidirectionalTree(PoESkillTree tree, string fromId, string toId)
   368→        {
   369→            var from = tree.Nodes.Find(n => n.Id == fromId);
   370→            var to = tree.Nodes.Find(n => n.Id == toId);
   371→            if (from == null || to == null) return;
   372→            if (!from.Connections.Contains(toId)) from.Connections.Add(toId);
   373→            if (!to.Connections.Contains(fromId)) to.Connections.Add(fromId);
   374→        }

   378→        // ====================================================================
   379→        private static void BuildAscendancyZone(PoESkillTree tree, float cx, float cy, BranchType branch)
   380→        {
   381→            // Nodo de entrada a la zona de Ascendancy
   382→            string ascendEntryId = "ascend-entry";
   383→            var ascendEntry = new PoESkillNode
   384→            {
   385→                Id = ascendEntryId, Name = "Ascendencia (Lv 100+)", Branch = "Ascendancy",
   386→                Type = NodeType.Keystone, Cost = 0,
   387→                Effect = "Desbloquea potenciadores procedurales. Genera nuevos nodos cada nivel.",
   388→                X = cx, Y = cy,
   389→            };
   390→            tree.Nodes.Add(ascendEntry);
   391→            // Conectar al nodo inicial (null-safe)
   392→            ConnectBidirectionalTree(tree, "start", ascendEntryId);
   393→
   394→            // 6 potenciadores procedurales base (se generan más al subir de nivel)
   395→            string[] ascendNames = {
   396→                "Potenciador Estelar I", "Potenciador Estelar II", "Potenciador Estelar III",
   397→                "Potenciador Cósmico I", "Potenciador Cósmico II", "Potenciador Cósmico III",
   398→            };
   399→            string[] ascendEffects = GetAscendancyEffects(branch);
   400→
   401→            string prevAscendId = ascendEntryId;
   402→            for (int i = 0; i < 6; i++)
   403→            {
   404→                float angle = i * 60f * (float)Math.PI / 180f;
   405→                string nodeId = $"ascend-{i}";
   406→                var node = new PoESkillNode
   407→                {
   408→                    Id = nodeId, Name = ascendNames[i], Branch = "Ascendancy",
   409→                    Type = NodeType.Ascendancy, Cost = 5,
   410→                    Effect = ascendEffects[i],
   411→                    X = cx + (float)Math.Cos(angle) * 200f,
   412→                    Y = cy + (float)Math.Sin(angle) * 200f,
   413→                };
   414→                node.Connections.Add(prevAscendId);
   415→                ConnectBidirectionalTree(tree, prevAscendId, nodeId);
   416→                tree.Nodes.Add(node);
   417→                prevAscendId = nodeId;
   418→            }
   419→
   420→            // KEYSTONE DE REGENERACIÓN DE SALUD (nivel 100+)
   421→            // Distancia y Melee: regenera 0.01% del daño causado como salud
   422→            // Magia: igual pero las invocaciones NO cuentan
   423→            string regenKeystoneId = "ascend-regen-keystone";
   424→            string regenEffect = branch == BranchType.Magic
   425→                ? "REGENERACIÓN VITAL: Regenera el 0.01% de tu salud máxima por cada punto de daño mágico causado. El daño de invocaciones NO cuenta para esta regeneración."
   426→                : "REGENERACIÓN VITAL: Regenera el 0.01% de tu salud máxima por cada punto de daño causado al enemigo. Lifesteal pasivo permanente.";
   427→
   428→            var regenKeystone = new PoESkillNode
   429→            {
   430→                Id = regenKeystoneId, Name = "Regeneración Vital (Keystone)",
   431→                Branch = "Ascendancy", Type = NodeType.Keystone, Cost = 10,
   432→                Effect = regenEffect,
   433→                X = cx + 100f, Y = cy + 250f,
   434→            };
   435→            regenKeystone.Connections.Add(prevAscendId);
   436→            ConnectBidirectionalTree(tree, prevAscendId, regenKeystoneId);
   437→            tree.Nodes.Add(regenKeystone);
   438→
   439→            // Potenciadores adicionales que se generan proceduralmente
   440→            // (el sistema generará más nodos en runtime cuando el jugador suba de nivel post-100)
   441→            for (int i = 6; i < 12; i++)
   442→            {
   443→                float angle = i * 60f * (float)Math.PI / 180f;
   444→                string nodeId = $"ascend-proc-{i}";
   445→                var node = new PoESkillNode
   446→                {
   447→                    Id = nodeId, Name = $"Potenciador Procedural {i-5}",
   448→                    Branch = "Ascendancy", Type = NodeType.Ascendancy, Cost = 5,
   449→                    Effect = GetProceduralEffect(branch, i),
   450→                    X = cx + (float)Math.Cos(angle) * 300f,
   451→                    Y = cy + (float)Math.Sin(angle) * 300f,
   452→                };
   453→                node.Connections.Add(prevAscendId);
   454→                ConnectBidirectionalTree(tree, prevAscendId, nodeId);
   455→                tree.Nodes.Add(node);
   456→                prevAscendId = nodeId;
   457→            }
   458→        }
