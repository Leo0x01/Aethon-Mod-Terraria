    70→export interface SkillTier {
    71→  min: number;
    72→  max: number;
    73→  perLevel: number;
    74→}
    75→
    76→export const SKILL_TIERS: SkillTier[] = [
    77→  { min: 1, max: 10, perLevel: 1 },
    78→  { min: 11, max: 20, perLevel: 2 },

    81→  { min: 41, max: 50, perLevel: 5 },
    82→  { min: 51, max: 60, perLevel: 6 },
    83→  { min: 61, max: 70, perLevel: 7 },
    84→  { min: 71, max: 80, perLevel: 8 },
    85→  { min: 81, max: 90, perLevel: 9 },
    86→  { min: 91, max: 100, perLevel: 10 },
    87→  { min: 101, max: Infinity, perLevel: 10 },
    88→];
    89→
    90→/** Total skill points earned once the weapon reaches `level` (1-indexed). */
    91→export function cumulativeSkillPoints(level: number): number {
    92→  if (level <= 0) return 0;
    93→  let total = 0;
    94→  for (const tier of SKILL_TIERS) {
    95→    const upper = Math.min(level, tier.max);
    96→    if (upper >= tier.min) {
    97→      total += (upper - tier.min + 1) * tier.perLevel;
    98→    }
    99→    if (level <= tier.max) break;
   100→  }
   101→  return total;
   102→}
   103→
   104→/** Skill points gained specifically at this level. */
   105→export function pointsForLevel(level: number): number {
   106→  for (const tier of SKILL_TIERS) {
   107→    if (level >= tier.min && level <= tier.max) return tier.perLevel;
   108→  }
   109→  return 10; // post-100 flat
   110→}
   111→
   112→/** XP required to go FROM `level` TO `level+1`. */
   113→export function xpForNextLevel(level: number): number {
   114→  return Math.floor(80 * Math.pow(level, 1.5));
   115→}
   116→
   117→/** Cumulative XP needed to REACH `level` (starting from level 1). */
   118→export function cumulativeXp(level: number): number {
   119→  let total = 0;
   120→  for (let l = 1; l < level; l++) total += xpForNextLevel(l);
   121→  return total;
   122→}

   144→// Helper to build constellation nodes on a hex grid per branch.
   145→function buildTree(
   146→  branchDefs: { name: string; nodes: { n: string; e: string; r?: NodeRarity; c?: number }[] }[],
   147→): SkillNode[] {
   148→  const nodes: SkillNode[] = [];
   149→  const branchCount = branchDefs.length;
   150→  branchDefs.forEach((branch, bi) => {
   151→    const angle = (bi / branchCount) * Math.PI * 2 - Math.PI / 2;
   152→    const baseR = 0.18;
   153→    branch.nodes.forEach((node, ni) => {
   154→      const r = baseR + ni * 0.13;
   155→      const jitter = (bi % 2 === 0 ? 0.04 : -0.04);
   156→      const x = 0.5 + Math.cos(angle + jitter * ni) * r * 1.15;
   157→      const y = 0.5 + Math.sin(angle + jitter * ni) * r * 1.0;
   158→      nodes.push({
   159→        id: `${branch.name}-${ni}`,
   160→        name: node.n,
   161→        branch: branch.name,
   162→        rarity: node.r ?? "common",
   163→        cost: node.c ?? 1,
   164→        effect: node.e,
   165→        prereq: ni > 0 ? `${branch.name}-${ni - 1}` : undefined,
   166→        x,
   167→        y,
   168→      });
   169→    });
   170→  });
   171→  return nodes;
   172→}
