    46→/**
    47→ * Deterministic pseudo-random from a seed (mulberry32).
    48→ */
    49→function mulberry32(seed: number) {
    50→  return function () {
    51→    let t = (seed += 0x6d2b79f5);
    52→    t = Math.imul(t ^ (t >>> 15), t | 1);
    53→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    54→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    55→  };
    56→}

    63→function readInitialBuild(): {
    64→  weaponId: WeaponId;
    65→  seed: number;
    66→  allocated: Set<string>;
    67→} {
    68→  if (typeof window === "undefined") {
    69→    return {
    70→      weaponId: "book" as WeaponId,
    71→      seed: 1337,
    72→      allocated: new Set<string>(),
    73→    };
    74→  }
    75→  // Decode using the weapon encoded IN the hash (its node list). We don't
    76→  // need to probe every weapon — the hash itself declares which weapon it's for.
    77→  const hash = window.location.hash;
    78→  if (!hash || !hash.startsWith("#v1.")) {
    79→    return {
    80→      weaponId: "book" as WeaponId,
    81→      seed: 1337,
    82→      allocated: new Set<string>(),
    83→    };
    84→  }
    85→  // Peek the weapon code to find the matching weapon definition.
    86→  const parts = hash.slice(4).split(".");
    87→  const wCode = parts[0];
    88→  const WEAPON_FROM_CODE: Record<string, WeaponId> = {
    89→    b: "bow",
    90→    s: "sword",
    91→    c: "cannon",
    92→    m: "book",
    93→  };
    94→  const wid = WEAPON_FROM_CODE[wCode];
    95→  if (!wid) {
    96→    return {
    97→      weaponId: "book" as WeaponId,
    98→      seed: 1337,
    99→      allocated: new Set<string>(),
   100→    };
   101→  }
   102→  const w = WEAPONS.find((x) => x.id === wid)!;
   103→  const ids = w.skillTree.map((n) => n.id);
   104→  const shared = decodeBuild(hash, ids);
   105→  if (!shared) {
   106→    return {
   107→      weaponId: "book" as WeaponId,
   108→      seed: 1337,
   109→      allocated: new Set<string>(),
   110→    };
   111→  }
   112→  const alloc = new Set<string>();
   113→  for (const idx of shared.nodeIndices) {
   114→    const id = w.skillTree[idx]?.id;
   115→    if (id) alloc.add(id);
   116→  }
   117→  return {
   118→    weaponId: shared.weaponId,
   119→    seed: shared.seed,
   120→    allocated: alloc,
   121→  };
   122→}

   124→export function SkillTreeView() {
   125→  // SSR-safe defaults; the shared build is restored in a mount effect below.
   126→  const [weaponId, setWeaponId] = useState<WeaponId>("book");
   127→  const [seed, setSeed] = useState(1337);
   128→  const [allocated, setAllocated] = useState<Set<string>>(new Set());
   129→  const [selected, setSelected] = useState<string | null>(null);
   130→  const [hydrated, setHydrated] = useState(false);
   131→  // Hovered node for the floating tooltip overlay (HTML, so text wraps nicely).
   132→  const [hovered, setHovered] = useState<string | null>(null);
   133→
   134→  // Node search: filters/highlights nodes in the constellation by name.
   135→  const [nodeQuery, setNodeQuery] = useState("");
   136→  const searchInputRef = useRef<HTMLInputElement | null>(null);
   137→
   138→  // Build import panel state (declared early so the keyboard handler can close it).
   139→  const [showImport, setShowImport] = useState(false);
   140→  const [importValue, setImportValue] = useState("");
   141→  const [importStatus, setImportStatus] = useState<
   142→    "idle" | "ok" | "error"
   143→  >("idle");
   144→
   145→  // Build comparison: snapshot "build A" to diff against the current build.
   146→  // Stored as a serialized hash string; persisted to localStorage so it
   147→  // survives reloads. null = no snapshot saved.
   148→  const SNAPSHOT_KEY = "aethon:build-snapshot-a";
   149→  const [snapshotA, setSnapshotA] = useState<string | null>(null);
   150→
   151→  const weapon = WEAPONS.find((w) => w.id === weaponId)!;
   152→
   153→  // Procedurally (re)place nodes using the seed → jitter + occasional extra branch arcs
   154→  const placed = useMemo<PlacedNode[]>(() => {
   155→    const rng = mulberry32(seed);
   156→    const W = 720;
   157→    const H = 620;
   158→    const cx = W / 2;
   159→    const cy = H / 2;
   160→    return weapon.skillTree.map((n) => {
   161→      const jx = (rng() - 0.5) * 0.08;
   162→      const jy = (rng() - 0.5) * 0.08;
   163→      const px = (n.x + jx) * W;
   164→      const py = (n.y + jy) * H;
   165→      // nudge toward center if too far out
   166→      const dx = px - cx;
   167→      const dy = py - cy;
   168→      const dist = Math.hypot(dx, dy);
   169→      const maxR = Math.min(W, H) / 2 - 40;
   170→      let fx = px;
   171→      let fy = py;
   172→      if (dist > maxR) {
   173→        fx = cx + (dx / dist) * maxR;
   174→        fy = cy + (dy / dist) * maxR;
   175→      }
   176→      return { ...n, px: fx, py: fy };
   177→    });
   178→  }, [weapon, seed]);
   179→
   180→  // reset allocations when weapon changes
   181→  const onWeaponChange = (id: WeaponId) => {
   182→    setWeaponId(id);
   183→    setAllocated(new Set());
   184→    setSelected(null);
   185→  };
   186→
   187→  const onReroll = () => {
   188→    setSeed(Math.floor(Math.random() * 1_000_000));
   189→    setAllocated(new Set());
   190→    setSelected(null);
   191→  };
   192→
   193→  // available points = a demo level (say 60) → cumulative
   194→  const demoLevel = 60;
   195→  const totalPoints = cumulativeSkillPoints(demoLevel);
   196→  const spent = useMemo(
   197→    () =>
   198→      placed
   199→        .filter((n) => allocated.has(n.id))
   200→        .reduce((sum, n) => sum + n.cost, 0),
   201→    [allocated, placed],
   202→  );
   203→  const available = totalPoints - spent;
   204→
   205→  // All node ids (in tree order) — used by build-share codec.
   206→  const allNodeIds = useMemo(() => weapon.skillTree.map((n) => n.id), [weapon]);
   207→
   208→  // Restore a shared build from the URL hash ONCE on mount (client-only,
   209→  // so SSR + hydration match — server renders empty defaults). This is a
   210→  // legitimate one-time hydration of browser-only state.
   211→  useEffect(() => {
   212→    /* eslint-disable react-hooks/set-state-in-effect */
   213→    const restored = readInitialBuild();
   214→    setWeaponId(restored.weaponId);
   215→    setSeed(restored.seed);
   216→    setAllocated(restored.allocated);
   217→    // Restore the build-compare snapshot from localStorage (if any).
   218→    setSnapshotA(loadString(SNAPSHOT_KEY));
   219→    setHydrated(true);
   220→    /* eslint-enable react-hooks/set-state-in-effect */
   221→  }, []);
   222→
   223→  // Listen for external build-import events (from BuildPresets or other
   224→  // components) to re-load the skill-tree state from the hash. This makes
   225→  // the skill tree reactive to one-click preset loads.
   226→  useEffect(() => {
   227→    const onImported = (e: Event) => {
   228→      const detail = (e as CustomEvent<{ hash: string }>).detail;
   229→      if (!detail?.hash) return;
   230→      for (const w of WEAPONS) {
   231→        const ids = w.skillTree.map((n) => n.id);
   232→        const shared = decodeBuild(detail.hash, ids);
   233→        if (shared && shared.weaponId === w.id) {
   234→          const alloc = new Set<string>();
   235→          for (const idx of shared.nodeIndices) {
   236→            const id = w.skillTree[idx]?.id;
   237→            if (id) alloc.add(id);
   238→          }
   239→          setWeaponId(shared.weaponId);
   240→          setSeed(shared.seed);
   241→          setAllocated(alloc);
   242→          setSelected(null);
   243→          break;
   244→        }
   245→      }
   246→    };
   247→    window.addEventListener("aethon:build-imported", onImported);
   248→    return () => window.removeEventListener("aethon:build-imported", onImported);
   249→  }, []);
   250→
   251→  // Global keyboard shortcuts:
   252→  //  - "/" focuses the node search box (unless already typing in an input)
   253→  //  - "Escape" blurs the focused node / clears the search / closes the import panel
   254→  useEffect(() => {
   255→    const onKeyDown = (e: KeyboardEvent) => {
   256→      const target = e.target as HTMLElement | null;
   257→      const isTyping =
   258→        target &&
   259→        (target.tagName === "INPUT" ||
   260→          target.tagName === "TEXTAREA" ||
   261→          target.isContentEditable);
   262→
   263→      if (e.key === "/" && !isTyping) {
   264→        // Only act when the skill-tree section is in view.
   265→        const section = document.querySelector("#skill-tree");
   266→        if (!section) return;
   267→        const rect = section.getBoundingClientRect();
   268→        if (rect.top < window.innerHeight && rect.bottom > 0) {
   269→          e.preventDefault();
   270→          searchInputRef.current?.focus();
   271→        }
   272→      } else if (e.key === "Escape") {
   273→        if (isTyping) {
   274→          // Clear the focused input's value if it's our search box.
   275→          if (target?.getAttribute("aria-label") === "Buscar nodo por nombre") {
   276→            setNodeQuery("");
   277→            (target as HTMLInputElement).blur();
   278→          }
   279→        } else {
   280→          // When not typing, Escape clears any active search query + closes
   281→          // the import panel + tooltips (if the skill-tree is in view).
   282→          const section = document.querySelector("#skill-tree");
   283→          if (section) {
   284→            const rect = section.getBoundingClientRect();
   285→            if (rect.top < window.innerHeight && rect.bottom > 0) {
   286→              if (nodeQuery) setNodeQuery("");
   287→              setShowImport(false);
   288→              setHovered(null);
   289→            }
   290→          }
   291→        }
   292→      }
   293→    };
   294→    window.addEventListener("keydown", onKeyDown);
   295→    return () => window.removeEventListener("keydown", onKeyDown);
   296→  }, [nodeQuery]);
   297→
   298→  // Persist current build to URL hash — but only AFTER hydration so we don't
   299→  // overwrite the shared hash we just restored with empty defaults. Preserves
   300→  // any codex loadout already in the hash so the two features coexist.
   301→  useEffect(() => {
   302→    if (!hydrated) return;
   303→    // Read existing codex segment from the hash (if any) to preserve it.
   304→    let codexIds: string[] | undefined;
   305→    const existingHash =
   306→      typeof window !== "undefined" ? window.location.hash : "";
   307→    if (existingHash) {
   308→      const shared = decodeBuild(existingHash, allNodeIds);
   309→      if (shared && shared.codexIndices.length > 0) {
   310→        codexIds = codexIdsFromIndices(
   311→          shared.weaponId,
   312→          shared.codexIndices,
   313→        );
   314→      }
   315→    }
   316→    const enc = encodeBuild(
   317→      weaponId,
   318→      seed,
   319→      [...allocated],
   320→      allNodeIds,
   321→      codexIds,
   322→    );
   323→    writeHashBuild(enc);
   324→  }, [allocated, weaponId, seed, allNodeIds, hydrated]);
   325→
   326→  const [shareStatus, setShareStatus] = useState<
   327→    "idle" | "copied" | "error"
   328→  >("idle");
   329→  const onShare = async () => {
   330→    // Preserve codex loadout from the existing hash so sharing includes it.
   331→    let codexIds: string[] | undefined;
   332→    const existingHash = window.location.hash;
   333→    if (existingHash) {
   334→      const shared = decodeBuild(existingHash, allNodeIds);
   335→      if (shared && shared.codexIndices.length > 0) {
   336→        codexIds = codexIdsFromIndices(shared.weaponId, shared.codexIndices);
   337→      }
   338→    }
   339→    const enc = encodeBuild(
   340→      weaponId,
   341→      seed,
   342→      [...allocated],
   343→      allNodeIds,
   344→      codexIds,
   345→    );
   346→    writeHashBuild(enc);
   347→    const ok = await copyShareUrl(enc);
   348→    setShareStatus(ok ? "copied" : "error");
   349→    window.setTimeout(() => setShareStatus("idle"), 2200);
   350→  };
   351→
   352→  // Build import: paste a share URL or hash to load someone else's build.
   353→  const onImport = () => {
   354→    // Accept either a full URL (...#v1.x.y.z / ...#v2.x.y.z.w) or a bare hash.
   355→    const raw = importValue.trim();
   356→    if (!raw) return;
   357→    let hash = "";
   358→    const v1Idx = raw.indexOf("#v1.");
   359→    const v2Idx = raw.indexOf("#v2.");
   360→    if (v1Idx >= 0) hash = raw.slice(v1Idx);
   361→    else if (v2Idx >= 0) hash = raw.slice(v2Idx);
   362→    else if (raw.startsWith("v1.") || raw.startsWith("v2.")) hash = `#${raw}`;
   363→    if (!hash) {
   364→      setImportStatus("error");
   365→      window.setTimeout(() => setImportStatus("idle"), 2400);
   366→      return;
   367→    }
   368→    // Decode against every weapon to find the matching one.
   369→    for (const w of WEAPONS) {
   370→      const ids = w.skillTree.map((n) => n.id);
   371→      const shared = decodeBuild(hash, ids);
   372→      if (shared && shared.weaponId === w.id) {
   373→        const alloc = new Set<string>();
   374→        for (const idx of shared.nodeIndices) {
   375→          const id = w.skillTree[idx]?.id;
   376→          if (id) alloc.add(id);
   377→        }
   378→        setWeaponId(shared.weaponId);
   379→        setSeed(shared.seed);
   380→        setAllocated(alloc);
   381→        setSelected(null);
   382→        writeHashBuild(hash);
   383→        // Notify the Memory Codex to sync its class + loadout to the imported build.
   384→        window.dispatchEvent(
   385→          new CustomEvent("aethon:build-imported", {
   386→            detail: { hash },
   387→          }),
   388→        );
   389→        setImportStatus("ok");
   390→        setImportValue("");
   391→        setShowImport(false);
   392→        window.setTimeout(() => setImportStatus("idle"), 2400);
   393→        return;
   394→      }
   395→    }
   396→    setImportStatus("error");
   397→    window.setTimeout(() => setImportStatus("idle"), 2400);
   398→  };
   399→
   400→  const onRandomize = () => {
   401→    // Greedy random allocation: pick random valid nodes until we run out of
   402→    // points or all eligible nodes are taken. Respects prereqs + budget.
   403→    const rng = mulberry32(Math.floor(Math.random() * 1_000_000));
   404→    const next = new Set<string>();
   405→    let remaining = totalPoints;
   406→    const order = [...weapon.skillTree];
   407→    const maxPasses = order.length;
   408→    for (let pass = 0; pass < maxPasses; pass++) {
   409→      const pool = [...order]
   410→        .map((n) => ({ n, r: rng() }))
   411→        .sort((a, b) => a.r - b.r)
   412→        .map((x) => x.n);
   413→      let progressed = false;
   414→      for (const n of pool) {
   415→        if (next.has(n.id)) continue;
   416→        if (n.cost > remaining) continue;
   417→        if (n.prereq && !next.has(n.prereq)) continue;
   418→        // 70% chance to take each eligible node → organic, not full, builds
   419→        if (rng() < 0.7) {
   420→          next.add(n.id);
   421→          remaining -= n.cost;
   422→          progressed = true;
   423→        }
   424→      }
   425→      if (!progressed || remaining <= 0) break;
   426→    }
   427→    setAllocated(next);
   428→    setSelected(null);
   429→  };
   430→
   431→  const canAllocate = (n: PlacedNode) => {
   432→    if (allocated.has(n.id)) return false;
   433→    if (available < n.cost) return false;
   434→    if (n.prereq && !allocated.has(n.prereq)) return false;
   435→    return true;
   436→  };
   437→
   438→  const toggle = (n: PlacedNode) => {
   439→    setSelected(n.id);
   440→    if (!canAllocate(n) && !allocated.has(n.id)) return;
   441→    setAllocated((prev) => {
   442→      const next = new Set(prev);
   443→      if (next.has(n.id)) {
   444→        // deallocate only if nothing depends on it
   445→        const hasDep = placed.some((p) => p.prereq === n.id && next.has(p.id));
   446→        if (hasDep) return next;
   447→        next.delete(n.id);
   448→      } else {
   449→        next.add(n.id);
   450→      }
   451→      return next;
   452→    });
   453→  };
   454→
   455→  const sel = placed.find((n) => n.id === selected) ?? null;
   456→
   457→  return (
   458→    <section id="skill-tree" className="relative py-24 sm:py-32">
   459→      <div className="mx-auto max-w-7xl px-4 sm:px-6">
   460→        <SectionHeading
   461→          kicker="Árbol de Habilidades"
   462→          title={
   463→            <>
   464→              Constelaciones{" "}
   465→              <span className="text-glow-violet">procedurales</span>
   466→            </>
   467→          }
   468→          subtitle="Cada arma genera su propio árbol con un seed. Las ramas irradian como estrellas conectadas por puentes de luz. Haz clic en un nodo para asignarlo; respeta prerrequisitos y tu presupuesto de puntos."
   469→        />
   470→
   471→        <div className="mt-10 flex flex-wrap items-center justify-center gap-2">
   472→          {WEAPONS.map((w) => (
   473→            <button
   474→              key={w.id}
   475→              onClick={() => onWeaponChange(w.id)}
   476→              className={cn(
   477→                "flex items-center gap-2 rounded-full border px-4 py-2 text-sm transition",
   478→                weaponId === w.id
   479→                  ? "border-transparent"
   480→                  : "border-border/60 text-muted-foreground hover:text-foreground",
   481→              )}
   482→              style={
   483→                weaponId === w.id
   484→                  ? {
   485→                      borderColor: w.accent,
   486→                      color: w.accent,
   487→                      boxShadow: `0 0 20px -6px ${w.accent}88`,
   488→                    }
   489→                  : undefined
   490→              }
   491→            >
   492→              <span
   493→                className="h-2 w-2 rounded-full"
   494→                style={{ background: w.accent }}
   495→              />
   496→              {w.name}
   497→            </button>
   498→          ))}
   499→        </div>
   500→
   501→        <div className="mt-8 grid grid-cols-1 gap-6 lg:grid-cols-[1.4fr_0.6fr]">
   502→          {/* CANVAS */}
   503→          <div className="relative overflow-hidden rounded-3xl border border-border/60 bg-card/30 p-2">
   504→            <div className="pointer-events-none absolute inset-0 bg-cosmic-grid opacity-25" />
   505→            <div className="pointer-events-none absolute inset-0">
   506→              <div className="absolute left-1/2 top-1/2 h-40 w-40 -translate-x-1/2 -translate-y-1/2 rounded-full bg-primary/10 blur-3xl" />
   507→            </div>
   508→
   509→            {/* node search (top-left) */}
   510→            <div className="absolute left-3 top-3 z-10 flex items-center gap-2">
   511→              <div className="relative">
   512→                <span className="pointer-events-none absolute left-2.5 top-1/2 -translate-y-1/2 text-muted-foreground">
   513→                  ⌕
   514→                </span>
   515→                <input
   516→                  ref={searchInputRef}
   517→                  value={nodeQuery}
   518→                  onChange={(e) => setNodeQuery(e.target.value)}
   519→                  placeholder="buscar nodo…  ( / )"
   520→                  className="w-36 rounded-full border border-border/60 bg-card/80 py-1.5 pl-7 pr-7 text-xs text-foreground backdrop-blur placeholder:text-muted-foreground/60 focus:border-primary/50 focus:outline-none focus:ring-1 focus:ring-primary/30 sm:w-48"
   521→                  aria-label="Buscar nodo por nombre"
   522→                />
   523→                {nodeQuery && (
   524→                  <button
   525→                    onClick={() => setNodeQuery("")}
   526→                    className="absolute right-1.5 top-1/2 -translate-y-1/2 text-muted-foreground transition hover:text-foreground"
   527→                    aria-label="Limpiar búsqueda"
   528→                  >
   529→                    ✕
   530→                  </button>
   531→                )}
   532→              </div>
   533→              {nodeQuery && (
   534→                <span className="hidden rounded-full bg-primary/10 px-2 py-1 font-mono text-[10px] text-primary sm:inline">
   535→                  {placed.filter((n) =>
   536→                    n.name.toLowerCase().includes(nodeQuery.toLowerCase()),
   537→                  ).length}{" "}
   538→                  / {placed.length}
   539→                </span>
   540→              )}
   541→            </div>
   542→
   543→            {/* controls */}
   544→            <div className="absolute right-3 top-3 z-10 flex flex-wrap justify-end gap-2">

   985→function BuildSummary({

  1077→                  {nodes.map((n) => (

  1231→/**
  1232→ * Build comparison panel: lets users snapshot the current build ("build A"),
  1233→ * then compare it against the live build. Highlights nodes that are:
  1234→ *  - added (in current, not in A) — green
  1235→ *  - removed (in A, not in current) — red
  1236→ *  - unchanged — dim
  1237→ * Only compares when the snapshot's weapon matches the current weapon.
  1238→ */
  1239→function BuildCompare({
  1240→  weapon,
  1241→  currentAllocated,
  1242→  placed,
  1243→  snapshotA,
  1244→  onSaveSnapshot,
  1245→  onClearSnapshot,
  1246→}: {
  1247→  weapon: (typeof WEAPONS)[number];
  1248→  currentAllocated: Set<string>;
  1249→  placed: PlacedNode[];
  1250→  snapshotA: string | null;
  1251→  onSaveSnapshot: () => void;
  1252→  onClearSnapshot: () => void;
  1253→}) {
  1254→  const allNodeIds = weapon.skillTree.map((n) => n.id);
  1255→
  1256→  const snapshotAllocated = useMemo(() => {
  1257→    if (!snapshotA) return null;
  1258→    const shared = decodeBuild(snapshotA, allNodeIds);
  1259→    if (!shared || shared.weaponId !== weapon.id) return null;
  1260→    const ids = new Set<string>();
  1261→    for (const idx of shared.nodeIndices) {
  1262→      const id = weapon.skillTree[idx]?.id;
  1263→      if (id) ids.add(id);
  1264→    }
  1265→    return ids;
  1266→  }, [snapshotA, allNodeIds, weapon]);
  1267→
  1268→  const diff = useMemo(() => {
  1269→    if (!snapshotAllocated) return null;
  1270→    const added: string[] = [];
  1271→    const removed: string[] = [];
  1272→    const unchanged: string[] = [];
  1273→    for (const n of placed) {
  1274→      const inCur = currentAllocated.has(n.id);
  1275→      const inSnap = snapshotAllocated.has(n.id);
  1276→      if (inCur && !inSnap) added.push(n.id);
  1277→      else if (!inCur && inSnap) removed.push(n.id);
  1278→      else if (inCur && inSnap) unchanged.push(n.id);
  1279→    }
  1280→    return { added, removed, unchanged: unchanged.length };
  1281→  }, [snapshotAllocated, currentAllocated, placed]);
  1282→
  1283→  const snapshotMeta = useMemo(() => {
  1284→    if (!snapshotA) return null;
  1285→    const shared = decodeBuild(snapshotA, allNodeIds);
  1286→    if (!shared) return null;
  1287→    const snapWeapon = WEAPONS.find((w) => w.id === shared.weaponId);
  1288→    return snapWeapon
  1289→      ? { name: snapWeapon.name, count: shared.nodeIndices.length }
  1290→      : null;
  1291→  }, [snapshotA, allNodeIds]);
  1292→
  1293→  const sameWeapon = snapshotAllocated !== null;
  1294→
  1295→  return (
  1296→    <motion.div
  1297→      initial={{ opacity: 0, y: 16 }}
  1298→      whileInView={{ opacity: 1, y: 0 }}
  1299→      viewport={{ once: true }}
  1300→      transition={{ duration: 0.4 }}
  1301→      className="glass-panel mt-6 rounded-3xl border border-border/60 p-6"
  1302→    >
  1303→      <div className="flex flex-wrap items-center justify-between gap-3">
  1304→        <div className="flex items-center gap-3">
  1305→          <span
  1306→            className="flex h-10 w-10 items-center justify-center rounded-xl text-lg"
  1307→            style={{
  1308→              background: weapon.accentSoft,
  1309→              color: weapon.accent,
  1310→            }}
  1311→          >
  1312→            ⚖
  1313→          </span>
  1314→          <div>
  1315→            <h3 className="text-base font-semibold">Comparar builds</h3>
  1316→            <p className="font-mono text-[10px] uppercase tracking-wider text-muted-foreground">
  1317→              Guarda una snapshot y compara contra la build actual
  1318→            </p>
  1319→          </div>
  1320→        </div>
  1321→        <div className="flex items-center gap-2">
  1322→          {!snapshotA ? (
  1323→            <button
  1324→              onClick={onSaveSnapshot}
  1325→              className="rounded-full border border-accent/40 bg-accent/10 px-4 py-1.5 text-xs font-semibold text-accent transition hover:border-accent hover:bg-accent/20"
  1326→            >
  1327→              📸 guardar build A
  1328→            </button>
  1329→          ) : (
  1330→            <>
  1331→              <span className="font-mono text-[10px] uppercase tracking-wider text-muted-foreground">
  1332→                A: {snapshotMeta?.name} · {snapshotMeta?.count} nodos
  1333→              </span>
  1334→              <button
  1335→                onClick={onSaveSnapshot}
  1336→                disabled={!sameWeapon}
  1337→                className="rounded-full border border-border/60 px-3 py-1.5 text-xs text-muted-foreground transition hover:border-accent/50 hover:text-accent disabled:opacity-40"
  1338→                title={
  1339→                  sameWeapon
  1340→                    ? "Sobrescribir snapshot A"
  1341→                    : "Cambia al arma de la snapshot para sobrescribir"
  1342→                }
  1343→              >
  1344→                ↻ actualizar
  1345→              </button>
  1346→              <button
  1347→                onClick={onClearSnapshot}
  1348→                className="rounded-full border border-destructive/40 px-3 py-1.5 text-xs text-destructive transition hover:bg-destructive/10"
  1349→              >
  1350→                ✕ borrar
  1351→              </button>
  1352→            </>
  1353→          )}
  1354→        </div>
  1355→      </div>
  1356→
  1357→      {snapshotA && !sameWeapon && (
  1358→        <div className="mt-4 rounded-xl border border-destructive/30 bg-destructive/5 p-3 text-xs text-muted-foreground">
  1359→          ⚠ La snapshot A es para{" "}
  1360→          <span className="text-foreground">
  1361→            {(() => {
  1362→              const shared = decodeBuild(snapshotA, allNodeIds);
  1363→              return shared ? WEAPONS.find((w) => w.id === shared.weaponId)?.name : "?";
  1364→            })()}
  1365→          </span>
  1366→          . Cambia a esa arma para comparar, o borra la snapshot.
  1367→        </div>
  1368→      )}
  1369→
  1370→      {diff && sameWeapon && (
  1371→        <div className="mt-4 grid grid-cols-1 gap-3 sm:grid-cols-3">
  1372→          <DiffStat
  1373→            label="Añadidos"
  1374→            count={diff.added.length}
  1375→            color="#7ee3c4"
  1376→            icon="+"
  1377→          />
  1378→          <DiffStat
  1379→            label="Quitados"
  1380→            count={diff.removed.length}
  1381→            color="#ff7a7a"
  1382→            icon="−"
  1383→          />
  1384→          <DiffStat
  1385→            label="Sin cambios"
  1386→            count={diff.unchanged}
  1387→            color="rgba(200,200,230,0.7)"
  1388→            icon="="
  1389→          />
  1390→        </div>
  1391→      )}
  1392→
  1393→      {diff && sameWeapon && (diff.added.length > 0 || diff.removed.length > 0) && (
  1394→        <div className="mt-4 space-y-2">
  1395→          {diff.added.length > 0 && (
  1396→            <DiffList
  1397→              title="Nodos añadidos (en actual, no en A)"
  1398→              ids={diff.added}
  1399→              placed={placed}
  1400→              color="#7ee3c4"
  1401→              sign="+"
  1402→            />
  1403→          )}
  1404→          {diff.removed.length > 0 && (
  1405→            <DiffList
  1406→              title="Nodos quitados (en A, no en actual)"
  1407→              ids={diff.removed}
  1408→              placed={placed}
  1409→              color="#ff7a7a"
  1410→              sign="−"
  1411→            />
  1412→          )}
  1413→        </div>
  1414→      )}
  1415→      {diff && sameWeapon && diff.added.length === 0 && diff.removed.length === 0 && (
  1416→        <p className="mt-4 rounded-xl border border-border/50 bg-card/30 p-3 text-center text-xs text-muted-foreground">
  1417→          ✓ Las builds son idénticas
  1418→        </p>
  1419→      )}
  1420→    </motion.div>
  1421→  );
  1422→}
