    91→            }
    92→            listSize = allBasePanel.Count;
    93→            NodeParent Node;
    94→            int state = 0; // disactivated and locked
    95→            for (int i = 0; i < listSize; i++)
    96→            {
    97→                Node = allBasePanel[i].node;
    98→                state = 0;
    99→                if (Node.GetEnable) //if the node is enabled
   100→                    state = 3;

   329→        public void DrawConnection(Color _color, Vector2 point1, Vector2 point2,NodeParent node,NodeParent neighboor)
   330→        {
   331→            float angle = 0;
   332→            float distance = 0;
   333→            //angle = (float)(Math.Atan2(point2.Y- point1.Y, point2.X - point1.X) *(180f/Math.PI));
   334→            angle = (float)(Math.Atan2(point2.Y - point1.Y, point2.X - point1.X) );
   335→            distance = (float)Math.Sqrt(Math.Pow(point2.X - point1.X, 2) + Math.Pow(point2.Y - point1.Y, 2));
   336→            Connection BG = new Connection(angle, distance, 10)
   337→            {
   338→                color = Color.DarkSlateGray
   339→                
   340→            };
   341→            
   342→            Connection connection = new Connection(angle, distance, 6)
   343→            {
   344→                color = _color,
   345→            };
   346→            BG.basePos = new Vector2(point1.X + SKILL_SIZE * 0.5f, point1.Y + SKILL_SIZE * 0.5f);
   347→            connection.basePos = new Vector2(point1.X+ SKILL_SIZE*0.5f, point1.Y+ SKILL_SIZE * 0.5f);
   348→            BG.bg = true;
   349→            BG.neighboor = neighboor;
   350→            BG.node = node;
   351→            connection.neighboor = neighboor;
   352→            connection.node = node;
   353→            allConnection.Add(BG);
   354→            allConnection.Add(connection);
   355→
   356→        }

   359→        private void ScrollUpDown(UIScrollWheelEvent evt, UIElement listeningElement)
   360→        {
   361→            Vector2 Center = new Vector2(Main.screenWidth * .5f, Main.screenHeight * .5f);
   362→            float preZoom = Zoom;
   363→            Vector2 mouseoffset = evt.MousePosition - Center;
   364→            
   365→            if (evt.ScrollWheelValue > 0)
   366→            {
   367→                Zoom = Mathf.Clamp(1.1f * Zoom, zoomMin, zoomMax);
   368→                
   369→            }
   370→            else if (evt.ScrollWheelValue < 0)
   371→            {
   372→                Zoom = Mathf.Clamp(0.85f * Zoom, zoomMin, zoomMax);
   373→                
   374→            }
   375→            float ratio = Zoom / preZoom;
   376→            offSet /= ratio;
   377→            if (ratio != 1) { 
   378→                if (evt.ScrollWheelValue > 0)
   379→                    offSet -= mouseoffset * Zoom * 0.1f;
   380→               else
   381→                    offSet += mouseoffset * Zoom * 0.02f;
   382→            }
   383→
   384→            Init();
   385→        }

   514→        private void OnClickNode(UIMouseEvent evt, UIElement listeningElement, NodeParent node)
   515→        {
   516→            if (node.GetActivate)
   517→            {
   518→                if (node.GetNodeType == NodeType.Class)
   519→                {
   520→                    node.ToggleEnable();
   521→                    
   522→                    UpdateValue();
   523→                    if (node.GetEnable == false)
   524→                        Terraria.Audio.SoundEngine.PlaySound(SoundID.MenuClose);
   525→                    else
   526→                        Terraria.Audio.SoundEngine.PlaySound(SoundID.MenuOpen);
   527→                    return;
   528→                }
   529→            }
   530→
   531→            RPGPlayer rPGPlayer = Main.player[Main.myPlayer].GetModPlayer<RPGPlayer>();
   532→            switch(node.CanUpgrade(rPGPlayer.GetSkillPoints, rPGPlayer.GetLevel()))
   533→            {
   534→                case Reason.CanUpgrade:
   535→                    rPGPlayer.SpentSkillPoints(0);
   536→                    node.Upgrade();
   537→                    UpdateToolTip(node);
   538→                    UpdateValue();
   539→                    Terraria.Audio.SoundEngine.PlaySound(SoundID.MenuOpen);
   540→                    break;
   541→                default:
   542→                    Terraria.Audio.SoundEngine.PlaySound(SoundID.MenuClose);
   543→                    break;
   544→            }
   545→

   548→        private void OpenToolTip(UIMouseEvent evt, UIElement listeningElement,NodeParent node)
   549→        {
   550→            if (node == null)
   551→            {
   552→                return;
   553→            }
   554→            toolTip = new UIPanel();
   555→            float unzoomMult = ScreenMult;
   556→
   557→            toolTip.Left.Set((node.menuPos.X + SKILL_SIZE*2 + offSet.X)* sizeMultplier, 0);
   558→            toolTip.Top.Set((node.menuPos.Y - SKILL_SIZE * 2 + offSet.Y)* sizeMultplier, 0);
   559→
   560→            float TTWidth = 500;
   561→            if (node.GetNodeType == NodeType.Class)
   562→            {
   563→                ClassType CT = (node.GetNode as ClassNode).GetClassType;
   564→                JsonChrClass ClassInfo = JsonCharacterClass.GetJsonCharList.GetClass(CT);
   565→                if (ClassInfo.ManaShield > 0)
   566→                    TTWidth = 750;
   567→                
   568→            }
   569→            toolTip.Width.Set(TTWidth * unzoomMult, 0);
   570→            toolTip.Height.Set(350* unzoomMult, 0);
   571→            toolTip.SetPadding(0);
   572→            toolTip.BackgroundColor = new Color(73, 94, 171, 150);
   573→
   574→            UIText Name = new UIText("",0.5f,true);
   575→            Name.Left.Set(10, 0);
   576→            Name.Top.Set(10, 0);
   577→            switch (node.GetNodeType) {
   578→                case NodeType.Class:
   579→                    Name.SetText("Class : "+(node.GetNode as ClassNode).GetClassType);
   580→                    break;
   581→                case NodeType.Perk:
   582→                    Name.SetText("Perk : " + (node.GetNode as PerkNode).GetPerk + " \nLevel : " + node.GetLevel + " / " + node.GetMaxLevel);
   583→                    break;
   584→                case NodeType.Immunity:
   585→                    Name.SetText("Class : " + (node.GetNode as ImmunityNode).GetImmunity);
   586→                    break;
   587→                case NodeType.Damage:
   588→                    Name.SetText("Bonus "+(node.GetNode as DamageNode).GetDamageType + " Damage\nLevel : " + node.GetLevel + " / " + node.GetMaxLevel);
   589→                    break;
   590→                case NodeType.Leech:
   591→                    Name.SetText("Bonus " + (node.GetNode as LeechNode).GetLeechType + " Leech\nLevel : " + node.GetLevel + " / " + node.GetMaxLevel);
   592→                    break;
   593→                case NodeType.Speed:
   594→                    Name.SetText("Bonus " + (node.GetNode as SpeedNode).GetDamageType + " Speed\nLevel : " + node.GetLevel + " / " + node.GetMaxLevel);
   595→                    break;
   596→                case NodeType.Stats:
   597→                    Name.SetText("Bonus " + (node.GetNode as StatNode).GetStatType + " Stats\nLevel : " + node.GetLevel + " / " + node.GetMaxLevel);
   598→                    break;
   599→                case NodeType.LimitBreak:
   600→                    Name.SetText("LIMIT BREAK : " + (node.GetNode as LimitBreakNode).LimitBreakType, unzoomMult, false);
   601→                    break;
   602→            }
   603→
   604→            UIText info = new UIText("Level Required : " + node.GetLevelRequirement +"\nCost " + node.GetCostPerLevel + " Points Per Level\n");
   605→            info.Left.Set(50* unzoomMult, 0);
   606→            info.Top.Set(100* unzoomMult, 0);
   607→
   608→            UIText description = new UIText(SkillInfo.GetDesc(node.GetNode));
   609→            description.Left.Set(50* unzoomMult, 0);
   610→            description.Top.Set(170* unzoomMult, 0);
   611→            
   612→            Terraria.Audio.SoundEngine.PlaySound(SoundID.MenuTick);
   613→
   614→            backGround.Append(toolTip);
   615→            toolTip.Append(Name);
   616→            toolTip.Append(info);
   617→            toolTip.Append(description);
   618→
   619→
   620→            backGround.Recalculate();
   621→        }
