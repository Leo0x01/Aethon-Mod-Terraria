   273→   273→        // HELPER: Construir un cluster completo (estilo PoE)
   274→   274→        // Cada cluster tiene: 3-5 small nodes → 1 notable → opcional keystone
   275→   275→        // ====================================================================
   276→   276→        private static void BuildCluster(
   277→   277→            PoESkillTree tree, string clusterId, string clusterName, string icon,
   278→   278→            string desc, bool hasKeystone, float angleDeg, float dist,
   279→   279→            float cx, float cy, BranchType branch)
   280→   280→        {
   281→   281→            float rad = angleDeg * (float)Math.PI / 180f;
   282→   282→            float clusterX = cx + (float)Math.Cos(rad) * dist;
   283→   283→            float clusterY = cy + (float)Math.Sin(rad) * dist;
   284→   284→
   285→   285→            // Helper local para añadir bidireccional (con null check)
   286→   286→            void ConnectBidirectional(string fromId, string toId)
   287→   287→            {
   288→   288→                var fromNode = tree.Nodes.Find(n => n.Id == fromId);
   289→   289→                var toNode = tree.Nodes.Find(n => n.Id == toId);
   290→   290→                if (fromNode != null && toNode != null)
   291→   291→                {
   292→   292→                    if (!fromNode.Connections.Contains(toId)) fromNode.Connections.Add(toId);
   293→   293→                    if (!toNode.Connections.Contains(fromId)) toNode.Connections.Add(fromId);
   294→   294→                }
   295→   295→            }
   296→   296→
   297→   297→            // Nodo de entrada del cluster (conectado al camino principal)
   298→   298→            string entryId = $"{clusterId}-entry";
   299→   299→            var entryNode = new PoESkillNode
   300→   300→            {
   301→   301→                Id = entryId, Name = $"{clusterName} (Entrada)", Branch = clusterName,
   302→   302→                Type = NodeType.Small, Cost = 1,
   303→   303→                Effect = GetSmallEffect(clusterId, branch, 0),
   304→   304→                X = clusterX - 40f, Y = clusterY,
   305→   305→            };
   306→   306→            tree.Nodes.Add(entryNode);
   307→   307→
   308→   308→            // 3 small nodes en línea
   309→   309→            string prevId = entryId;
   310→   310→            for (int i = 0; i < 3; i++)
   311→   311→            {
   312→   312→                string nodeId = $"{clusterId}-small-{i}";
   313→   313→                float nx = clusterX + (i - 1) * 40f;
   314→   314→                float ny = clusterY + 30f;
   315→   315→                var node = new PoESkillNode
   316→   316→                {
   317→   317→                    Id = nodeId, Name = $"{clusterName} (Nodo {i+1})", Branch = clusterName,
   318→   318→                    Type = NodeType.Small, Cost = 1,
   319→   319→                    Effect = GetSmallEffect(clusterId, branch, i + 1),
   320→   320→                    X = nx, Y = ny,
   321→   321→                };
   322→   322→                node.Connections.Add(prevId);
   323→   323→                ConnectBidirectional(prevId, nodeId);
   324→   324→                tree.Nodes.Add(node);
   325→   325→                prevId = nodeId;
   326→   326→            }
   327→   327→
   328→   328→            // Notable del cluster
   329→   329→            string notableId = $"{clusterId}-notable";
   330→   330→            var notable = new PoESkillNode
   331→   331→            {
   332→   332→                Id = notableId, Name = GetNotableName(clusterId, branch),
   333→   333→                Branch = clusterName, Type = NodeType.Notable, Cost = 2,
   334→   334→                Effect = GetNotableEffect(clusterId, branch),
   335→   335→                X = clusterX + 20f, Y = clusterY + 80f,
   336→   336→            };
   337→   337→            notable.Connections.Add(prevId);
   338→   338→            ConnectBidirectional(prevId, notableId);
   339→   339→            tree.Nodes.Add(notable);
   340→   340→
   341→   341→            // Keystone del cluster (opcional)
   342→   342→            if (hasKeystone)
   343→   343→            {
   344→   344→                string keystoneId = $"{clusterId}-keystone";
   345→   345→                var keystone = new PoESkillNode
   346→   346→                {
   347→   347→                    Id = keystoneId, Name = GetKeystoneName(clusterId, branch),
   348→   348→                    Branch = clusterName, Type = NodeType.Keystone, Cost = 3,
   349→   349→                    Effect = GetKeystoneEffect(clusterId, branch),
   350→   350→                    X = clusterX + 60f, Y = clusterY + 130f,
   351→   351→                };
   352→   352→                keystone.Connections.Add(notableId);
   353→   353→                ConnectBidirectional(notableId, keystoneId);
   354→   354→                tree.Nodes.Add(keystone);
   355→   355→            }
   356→   356→        }
   357→   357→
   358→   358→        // Conectar dos clusters (camino entre ellos)
   359→   359→        private static void ConnectClusters(PoESkillTree tree, string fromCluster, string toCluster)
   360→   360→        {
   361→   361→            string fromId = $"{fromCluster}-entry";
   362→   362→            string toId = $"{toCluster}-entry";
   363→   363→            ConnectBidirectionalTree(tree, fromId, toId);
   364→   364→        }

   366→   366→        /// <summary>Conecta dos nodos bidireccionalmente (null-safe, evita duplicados).</summary>
   367→   367→        private static void ConnectBidirectionalTree(PoESkillTree tree, string fromId, string toId)
   368→   368→        {
   369→   369→            var from = tree.Nodes.Find(n => n.Id == fromId);
   370→   370→            var to = tree.Nodes.Find(n => n.Id == toId);
   371→   371→            if (from == null || to == null) return;
   372→   372→            if (!from.Connections.Contains(toId)) from.Connections.Add(toId);
   373→   373→            if (!to.Connections.Contains(fromId)) to.Connections.Add(fromId);
   374→   374→        }
